cat << 'EOF' > package.json
{
  "name": "mern-api",
  "version": "1.0.0",
  "type": "module",
  "main": "server.js",
  "scripts": {
    "start": "node server.js"
  },
  "dependencies": {
    "express": "^4.19.2"
  }
}
EOF

cat << 'EOF' > server.js
import express from 'express';

const app = express();
const PORT = process.env.PORT || 5000;

app.use(express.json());

// Root endpoint
app.get('/', (req, res) => {
  res.status(200).json({
    status: 'success',
    framework: 'Node.js Express (ES6 Modules)',
    message: 'Welcome to MERN API - Running via DooD Jenkins Pipeline!'
  });
});

// Health check endpoint
app.get('/health', (req, res) => {
  res.status(200).json({
    status: 'UP',
    uptime: process.uptime(),
    timestamp: new Date().toISOString()
  });
});

app.listen(PORT, '0.0.0.0', () => {
  console.log(`Server is running on port ${PORT}`);
});
EOF

FROM node:18-alpine

WORKDIR /app

COPY package*.json ./
RUN npm install --production

COPY . .

EXPOSE 5000

CMD ["npm", "start"]
EOF

cat << 'EOF' > Jenkinsfile
pipeline {
    agent any

    environment {
        IMAGE_NAME = 'mern-api'
        IMAGE_TAG = "${env.BUILD_NUMBER}"
        CONTAINER_NAME = "mern-api-test-${env.BUILD_NUMBER}"
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build Image') {
            steps {
                sh "docker build -t ${IMAGE_NAME}:${IMAGE_TAG} ."
            }
        }

        stage('Test Endpoints') {
            steps {
                script {
                    // Sibling container run via DooD
                    sh "docker run -d --name ${CONTAINER_NAME} -p 5000:5000 ${IMAGE_NAME}:${IMAGE_TAG}"
                    sleep 3
                    
                    // Testing ES6 API Endpoints
                    sh "curl -f http://localhost:5000/health"
                    sh "curl -f http://localhost:5000/"
                }
            }
        }
    }

    post {
        always {
            // Clean up sibling container & image from host
            sh "docker rm -f ${CONTAINER_NAME} || true"
            sh "docker rmi -f ${IMAGE_NAME}:${IMAGE_TAG} || true"
        }
    }
}
EOF
